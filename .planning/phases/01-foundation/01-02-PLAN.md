    import { ipcMain, app } from 'electron'
    import { AppQuitSchema, AppGetVersionResult } from '../../shared/validators/app'
    import { IPC_CHANNELS } from '../../shared/types/ipc'

    export function registerAppIPC(): void {
      ipcMain.handle(IPC_CHANNELS.APP_QUIT, async (_event, payload) => {
        const params = AppQuitSchema.parse(payload)
        try {
          if (params.force) {
            app.exit(0)
          } else {
            app.quit()
          }
          return { success: true }
        } catch (error) {
          return {
            success: false,
            error: error instanceof Error ? error.message : 'Unknown error',
          }
        }
      })

      ipcMain.handle(IPC_CHANNELS.APP_GET_VERSION, async () => {
        return app.getVersion()
      })
    }
    ```

    }

    export class ViewerRegistry {
      private viewers = new Map<string, ViewerDefinition>()

      register(type: string, def: ViewerDefinition): void {
        this.viewers.set(type, def)
      }

      getViewer(type: string): ViewerDefinition | undefined {
        return this.viewers.get(type)
      }

      render(type: string, props: ArtifactViewerProps): ReactNode {
        const def = this.viewers.get(type)
        if (!def) {
          return (
            <div className="flex items-center justify-center h-full p-4 text-text-faint">
              Unsupported artifact type: {type}
            </div>
          )
        }
        const SkeletonComponent = def.Skeleton as FC<ArtifactViewerProps>
        return (
          <Suspense fallback={<SkeletonComponent {...props} />}>
            <def.component {...props} />
          </Suspense>
        )
      }
    }

    export const viewerRegistry = new ViewerRegistry()
    ```
